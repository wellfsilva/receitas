Conectar com Google Planilhas

Deixe em branco para continuar usando o modo de testes local.

Passo a Passo de Configuração:

  1. Crie uma nova Planilha do Google. Na primeira linha (cabeçalho), escreva:
    id | nome | tipo | preaquecer | ingredientes | preparo | link
  2. No menu da planilha, acesse: Extensões > Apps Script.
  3. Apague todo o código existente lá e cole o código abaixo.
  4. Clique em Implantar > Nova implantação.
  5. Escolha o tipo App da Web. Em Quem pode acessar, selecione Qualquer pessoa (necessário para a requisição via navegador).
  6. Copie a URL do app da Web gerada e cole no campo acima!
Code.gs 
function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var recipes = [];
  
  // Pula a linha do cabeçalho
  for (var i = 1; i < data.length; i++) {
    if(!data[i][0]) continue;
    recipes.push({
      id: data[i][0],
      nome: data[i][1],
      tipo: data[i][2],
      preaquecer: data[i][3] === true || data[i][3] === "TRUE",
      ingredientes: JSON.parse(data[i][4] || "[]"),
      preparo: data[i][5],
      link: data[i][6]
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify(recipes))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var payload = JSON.parse(e.postData.contents);
    
    var newId = "rec_" + new Date().getTime();
    
    sheet.appendRow([
      newId,
      payload.nome,
      payload.tipo,
      payload.preaquecer,
      JSON.stringify(payload.ingredientes),
      payload.preparo,
      payload.link
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", id: newId }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
